##########################################################################
# zz_bulk_resettle - execution effects (v2.1: 补岗迁移 / 超额迁移 / 完全迁移)
#
# Scope convention: every decision effect runs with root = the planet the
# player clicked on (DESTINATION). "平民" (civilian pops) = pop categories
# flagged is_civilian_job = yes: civilian + maintenance_drone.
#
# Fee model (matches vanilla manual resettlement, per the in-game tooltip):
#   energy = 100 / 100 pops, unity = 10 / 100 pops
#   -> 100-pop chunk costs 100 energy + 10 unity;
#      single pop costs 1 energy + 0.1 unity.
# (Cost-multiplier modifiers such as edicts/authorities are not applied.)
#
# Notes:
#  - 超额迁移 uses ordered_owned_colony with a numeric scripted trigger to
#    select the source colony containing the most civilian-type pops.
##########################################################################

# ============================================================
# Phase: fill open jobs on root with civilians from any source
# Uses variable zz_br_need (exported by the caller).
# ============================================================
zz_br_fill_jobs_phase_effect = {
	# ---------- full 100-pop chunks (100 energy + 10 unity each) ----------
	while = {
		limit = {
			check_variable = {
				which = zz_br_need
				value >= 100
			}
			root = {
				owner = {
					has_resource = {
						type = energy
						amount >= 100
					}
					has_resource = {
						type = unity
						amount >= 10
					}
					any_owned_colony = {
						NOT = { is_same_value = root }
						has_orbital_bombardment = no
						has_ground_combat = no
						any_owned_pop_group = {
							OR = {
								is_pop_category = civilian
								is_pop_category = maintenance_drone
							}
							pop_group_size >= 100
						}
					}
				}
			}
		}
		count = 3000

		owner = {
			random_owned_colony = {
				limit = {
					NOT = { is_same_value = root }
					has_orbital_bombardment = no
					has_ground_combat = no
					any_owned_pop_group = {
						OR = {
							is_pop_category = civilian
							is_pop_category = maintenance_drone
						}
						pop_group_size >= 100
					}
				}
				random_owned_pop_group = {
					limit = {
						OR = {
							is_pop_category = civilian
							is_pop_category = maintenance_drone
						}
						pop_group_size >= 100
					}
					resettle_pop_group = {
						POP_GROUP = this
						PLANET = root
						AMOUNT = 100
					}
					root = {
						owner = {
							add_resource = {
								energy = -100
							}
							add_resource = {
								unity = -10
							}
						}
						subtract_variable = {
							which = zz_br_need
							value = 100
						}
						set_variable = {
							which = zz_br_transferred
							value = 1
						}
					}
				}
			}
		}
	}

	# ---------- remaining <100 pops one by one (1 energy + 0.1 unity each) ----------
	while = {
		limit = {
			check_variable = {
				which = zz_br_need
				value > 0
			}
			root = {
				owner = {
					has_resource = {
						type = energy
						amount >= 1
					}
					has_resource = {
						type = unity
						amount >= 0.1
					}
					any_owned_colony = {
						NOT = { is_same_value = root }
						has_orbital_bombardment = no
						has_ground_combat = no
						any_owned_pop_group = {
							OR = {
								is_pop_category = civilian
								is_pop_category = maintenance_drone
							}
						}
					}
				}
			}
		}
		count = 2000

		owner = {
			random_owned_colony = {
				limit = {
					NOT = { is_same_value = root }
					has_orbital_bombardment = no
					has_ground_combat = no
					any_owned_pop_group = {
						OR = {
							is_pop_category = civilian
							is_pop_category = maintenance_drone
						}
					}
				}
				random_owned_pop_group = {
					limit = {
						OR = {
							is_pop_category = civilian
							is_pop_category = maintenance_drone
						}
					}
					resettle_pop_group = {
						POP_GROUP = this
						PLANET = root
						AMOUNT = 1
					}
					root = {
						owner = {
							add_resource = {
								energy = -1
							}
							add_resource = {
								unity = -0.1
							}
						}
						subtract_variable = {
							which = zz_br_need
							value = 1
						}
						set_variable = {
							which = zz_br_transferred
							value = 1
						}
					}
				}
			}
		}
	}
}

# ============================================================
# Drain ALL civilian pops from event_target:zz_br_src into root,
# regardless of jobs/housing on root (overfill allowed).
# ============================================================
zz_br_drain_src_planet_effect = {
	event_target:zz_br_src = {
		# ---------- 100-pop chunks (100 energy + 10 unity each) ----------
		while = {
			limit = {
				any_owned_pop_group = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
					pop_group_size >= 100
				}
				root = {
					owner = {
						has_resource = {
							type = energy
							amount >= 100
						}
						has_resource = {
							type = unity
							amount >= 10
						}
					}
				}
			}
			count = 3000

			random_owned_pop_group = {
				limit = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
					pop_group_size >= 100
				}
				resettle_pop_group = {
					POP_GROUP = this
					PLANET = root
					AMOUNT = 100
				}
				root = {
					owner = {
						add_resource = {
							energy = -100
						}
						add_resource = {
							unity = -10
						}
					}
					set_variable = {
						which = zz_br_transferred
						value = 1
					}
				}
			}
		}

		# ---------- remainder / small groups, one pop at a time ----------
		while = {
			limit = {
				any_owned_pop_group = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
				}
				root = {
					owner = {
						has_resource = {
							type = energy
							amount >= 1

							type = unity
							amount >= 0.1
						}
					}
				}
			}
			count = 20000

			random_owned_pop_group = {
				limit = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
				}
				resettle_pop_group = {
					POP_GROUP = this
					PLANET = root
					AMOUNT = 1
				}
				root = {
					owner = {
						add_resource = {
							energy = -1
						}
						add_resource = {
							unity = -0.1
						}
					}
					set_variable = {

				}
			}
		}
	}

# ============================================================
# Decision effect: 补岗迁移 (fill jobs)
# ============================================================
zz_br_do_fill_jobs = {
	hidden_effect = {
		export_trigger_value_to_variable = {
			trigger = free_jobs
			variable = zz_br_need
		}
		set_variable = {
			which = zz_br_transferred
			value = 0
		}

		if = {
			limit = {
				check_variable = {
					which = zz_br_need
					value > 0
				}
			}
			zz_br_fill_jobs_phase_effect = yes
		}

		else = {
			owner = {
				country_event = {
					id = zz_br.0
				}
			}
		}

		# Pop transfers do not always trigger an immediate job evaluation.
		# Refresh employment before deciding whether every vacancy was filled.
		check_planet_employment = yes

		# outcome
		if = {
			limit = {
				free_jobs <= 0
			}
			owner = {
				country_event = {
					id = zz_br.1
				}
			}
		}
		else_if = {
			limit = {
				root = {
					owner = {
						NOR = {
							has_resource = {
								type = energy
								amount >= 1
							}
							has_resource = {
								type = unity
								amount >= 0.1
							}
						}
					}
				}
			}
			owner = {
				country_event = {
					id = zz_br.3
				}
			}
		}
		else_if = {
			limit = {
				root = {
					owner = {
						NOT = {
							any_owned_colony = {
								NOT = { is_same_value = root }
								any_owned_pop_group = {
									OR = {
										is_pop_category = civilian
										is_pop_category = maintenance_drone
									}
								}
							}
						}
					}
				}
			}
			owner = {
				country_event = {
					id = zz_br.2
				}
			}
		}
		else = {
			owner = {
				country_event = {
					id = zz_br.4
				}
			}
		}

		clear_variable = zz_br_need
		clear_variable = zz_br_transferred
	}
}

# ============================================================
# Decision effect: 超额迁移 = 补岗迁移 + drain all civilians of the
# source colony with the most civilian-type pops.
# ============================================================
zz_br_do_excess = {
	hidden_effect = {
		export_trigger_value_to_variable = {
			trigger = free_jobs
			variable = zz_br_need
		}
		set_variable = {
			which = zz_br_transferred
			value = 0
		}

		# 1) fill jobs exactly like 补岗迁移
		if = {
			limit = {
				check_variable = {
					which = zz_br_need
					value > 0
				}
			}
			zz_br_fill_jobs_phase_effect = yes
		}
		check_planet_employment = yes

		# 2) extra: pick the source colony with the most civilian-type pops,
		#    then move ALL of its civilians. ordered_owned_colony is supported
		#    by the 4.5 engine; weighted_random_owned_colony is not.
		if = {
			limit = {
				root = {
					owner = {
						any_owned_colony = {
							NOT = { is_same_value = root }
							has_orbital_bombardment = no
							has_ground_combat = no
							any_owned_pop_group = {
								OR = {
									is_pop_category = civilian
									is_pop_category = maintenance_drone
								}
							}
						}
					}
				}
			}
			owner = {
				ordered_owned_colony = {
					limit = {
						NOT = { is_same_value = root }
						has_orbital_bombardment = no
						has_ground_combat = no
						any_owned_pop_group = {
							OR = {
								is_pop_category = civilian
								is_pop_category = maintenance_drone
							}
						}
					}
					position = 0
					order_by = trigger:zz_br_civilian_pop_amount
					save_event_target_as = zz_br_src
				}
			}
			zz_br_drain_src_planet_effect = yes
		}

		# outcome
		if = {
			limit = {
				check_variable = {
					which = zz_br_transferred
					value = 0
				}
			}
			owner = {
				country_event = {
					id = zz_br.12
				}
			}
		}
		else_if = {
			limit = {
				root = {
					owner = {
						NOR = {
							has_resource = {
								type = energy
								amount >= 1
							}
							has_resource = {
								type = unity
								amount >= 0.1
							}
						}
					}
				}
			}
			owner = {
				country_event = {
					id = zz_br.13
				}
			}
		}
		else = {
			owner = {
				country_event = {
					id = zz_br.11
				}
			}
		}

		clear_variable = zz_br_need
		clear_variable = zz_br_transferred
	}
}

# ============================================================
# Decision effect: 完全迁移 = move ALL civilian pops of every other
# planet into root (no job/housing cap; fees paid until funds run out).
# ============================================================
zz_br_do_full = {
	hidden_effect = {
		set_variable = {
			which = zz_br_transferred
			value = 0
		}

		owner = {
			# repeat until no source planet still holds civilians
			while = {
				limit = {
					any_owned_colony = {
						NOT = { is_same_value = root }
						has_orbital_bombardment = no
						has_ground_combat = no
						any_owned_pop_group = {
							OR = {
								is_pop_category = civilian
								is_pop_category = maintenance_drone
							}
						}
					}
					root = {
						owner = {
							has_resource = {
								type = energy
								amount >= 1
							}
							has_resource = {
								type = unity
								amount >= 0.1
							}
						}
					}
				}
				count = 500

				random_owned_colony = {
					limit = {
						NOT = { is_same_value = root }
						has_orbital_bombardment = no
						has_ground_combat = no
						any_owned_pop_group = {
							OR = {
								is_pop_category = civilian
								is_pop_category = maintenance_drone
							}
						}
					}
					save_event_target_as = zz_br_src
				}
				zz_br_drain_src_planet_effect = yes
			}
		}

		# outcome
		if = {
			limit = {
				check_variable = {
					which = zz_br_transferred
					value = 0
				}
			}
			owner = {
				country_event = {
					id = zz_br.22
				}
			}
		}
		else_if = {
			limit = {
				root = {
					owner = {
						any_owned_colony = {
							NOT = { is_same_value = root }
							any_owned_pop_group = {
								OR = {
									is_pop_category = civilian
									is_pop_category = maintenance_drone
								}
							}
						}
					}
				}
			}
			owner = {
				country_event = {
					id = zz_br.23
				}
			}
		}
		else = {
			owner = {
				country_event = {
					id = zz_br.21
				}
			}
		}

		clear_variable = zz_br_transferred
	}
}
